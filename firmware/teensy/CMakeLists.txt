cmake_minimum_required(VERSION 3.16)
project(STUDIO LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(APPLE)
 execute_process(COMMAND xcrun --show-sdk-path OUTPUT_VARIABLE SDK OUTPUT_STRIP_TRAILING_WHITESPACE)
 include_directories(SYSTEM "${SDK}/usr/include/c++/v1")
endif()
file(GLOB CORE CONFIGURE_DEPENDS firmware/STUDIO/src/studio/*.cpp)
add_library(studio ${CORE} desktop/Host.cpp)
target_include_directories(studio PUBLIC firmware/STUDIO/src desktop)
target_compile_options(studio PRIVATE -Wall -Wextra -Wpedantic)
add_executable(studio_sim desktop/main.cpp)
target_link_libraries(studio_sim studio)
add_executable(studio_tests tests/tests.cpp)
target_link_libraries(studio_tests studio)
enable_testing()
add_test(NAME studio_tests COMMAND studio_tests)
